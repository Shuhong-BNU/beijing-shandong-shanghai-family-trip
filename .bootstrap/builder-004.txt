aily first/major nodes
    allnodes=[]; allmodes=[]
    last=None
    for d in p['days']:
        nodes=d[2]; modes=d[3]
        for i,n in enumerate(nodes):
            k=n[0]
            if last!=k:
                allnodes.append(n)
                if last is not None:
                    # segment mode: if internal day use previous local mode else flight/rail/taxi approximate
                    m=modes[i-1] if i>0 and i-1 < len(modes) else 'taxi'
                    allmodes.append(m)
                last=k
    mapdata[f'plan-{pid}']={'nodes':allnodes,'modes':allmodes,'plan':pid,'title':p['name']}
    for di,d in enumerate(p['days'],1):
        mapdata[f'{pid}-d{di}']={'nodes':d[2],'modes':d[3],'plan':pid,'title':f'{d[0]} {d[1]}'}
